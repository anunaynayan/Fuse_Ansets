"use client";

import { useTheme } from "@mui/material/styles";
import {
  LineChart,
  Line,
  XAxis,
  YAxis,
  CartesianGrid,
  Tooltip,
  ResponsiveContainer,
  Legend,
  Area,
  LabelList,
} from "recharts";
import { Card, Typography, CircularProgress } from "@mui/material";
import { useEffect, useState } from "react";

interface LineDataset {
  name: string;
  dataKey: string;
  color: string;
  strokeWidth?: number;
}

interface ChartPoint {
  [key: string]: string | number;
}

interface LineChartComponentProps {
  title?: string;
  dataUrl: string;
  datasets: LineDataset[];
  showGrid?: boolean;
  height?: number;
  smooth?: boolean;
  showDots?: boolean;
  animate?: boolean;
  area?: boolean;
  gradient?: boolean;
  showValueLabels?: boolean;
}

export default function LineChartComponent({
  title = "Line Chart",
  dataUrl,
  datasets,
  showGrid = true,
  height = 350,
  smooth = true,
  showDots = true,
  animate = true,
  area = false,
  gradient = false,
  showValueLabels = false,
}: LineChartComponentProps) {
  const theme = useTheme();
  const [data, setData] = useState<ChartPoint[] | null>(null);

  useEffect(() => {
    fetch(dataUrl)
      .then((res) => res.json())
      .then((res) => {
        // small timeout for nicer entry animation
        setTimeout(() => setData(res), 200);
      });
  }, [dataUrl]);

  if (!data)
    return (
      <Card sx={{ p: 3, textAlign: "center" }}>
        <CircularProgress />
        <Typography sx={{ mt: 1 }}>Loading chart...</Typography>
      </Card>
    );

  // animationDuration: 0 disables animation in Recharts
  const animationDuration = animate ? 1000 : 0;
  const lineType = smooth ? "monotone" : "linear";

  return (
    <div className="p-3">
      <Typography variant="h6" gutterBottom>
        {title}
      </Typography>

      <ResponsiveContainer width="100%" height={height}>
        <LineChart data={data} margin={{ top: 20, right: 25, left: 10, bottom: 10 }}>
          {showGrid && (
            <CartesianGrid strokeDasharray="3 3" stroke={theme.palette.divider} opacity={0.3} />
          )}

          <XAxis dataKey="name" stroke={theme.palette.text.secondary} tick={{ fontSize: 12 }} />
          <YAxis stroke={theme.palette.text.secondary} tick={{ fontSize: 12 }} />

          <Tooltip
            contentStyle={{
              background: theme.palette.background.paper,
              border: `1px solid ${theme.palette.divider}`,
            }}
            labelStyle={{
              color: theme.palette.text.primary,
              fontWeight: 600,
            }}
          />
          <Legend />

          {/* Gradient defs */}
          {gradient && (
            <defs>
              {datasets.map((ds) => (
                <linearGradient key={ds.dataKey} id={`grad-${ds.dataKey}`} x1="0" y1="0" x2="0" y2="1">
                  <stop offset="0%" stopColor={ds.color} stopOpacity={0.9} />
                  <stop offset="100%" stopColor={ds.color} stopOpacity={0.2} />
                </linearGradient>
              ))}
            </defs>
          )}

          {/* Optionally render area fills */}
          {area &&
            datasets.map((ds) => (
              <Area
                key={`area-${ds.dataKey}`}
                type={lineType as any}
                dataKey={ds.dataKey}
                stroke={ds.color}
                fill={gradient ? `url(#grad-${ds.dataKey})` : ds.color}
                fillOpacity={0.12}
                isAnimationActive={!!animationDuration}
                animationDuration={animationDuration}
              />
            ))}

          {datasets.map((line) => (
            <Line
              key={line.dataKey}
              type={lineType as any}
              dataKey={line.dataKey}
              stroke={gradient ? `url(#grad-${line.dataKey})` : line.color}
              strokeWidth={line.strokeWidth ?? 2}
              dot={showDots ? { r: 3 } : false}
              activeDot={showDots ? { r: 5 } : false}
              isAnimationActive={!!animationDuration}
              animationDuration={animationDuration}
            >
              {showValueLabels && (
                // LabelList works with Line to show point labels
                <LabelList dataKey={line.dataKey} position="top" style={{ fill: theme.palette.text.primary, fontSize: 12, fontWeight: 600 }} />
              )}
            </Line>
          ))}
        </LineChart>
      </ResponsiveContainer>
    </div>
  );
}
